WITH contract AS (
    SELECT id, user_id, contract_no, create_date as contract_date
    FROM contract
),
active_device AS (
    SELECT id, building_id, contract_id, serial, device_name
    FROM device
    WHERE device_type = 'ABT' AND linked_yn = 'Y'
),
latest_user_app_info_cte AS (
    SELECT login_id, app_version, update_date as app_update_date
    FROM (
        SELECT login_id, app_version, update_date,
               ROW_NUMBER() OVER (PARTITION BY login_id ORDER BY update_date DESC) AS rn
        FROM history_user_app_info
    ) t
    WHERE t.rn = 1
),
latest_so_version AS (
    SELECT device_id, version
    FROM (
        SELECT device_id, version,
               ROW_NUMBER() OVER (PARTITION BY device_id ORDER BY create_date DESC) as rn
        FROM statistics_operation
    ) t
    WHERE t.rn = 1
)

SELECT
    u.login_id,
    c.contract_date,
    ad.serial,
    ad.device_name,
    app.app_version,
    so.version AS op_version,
    COALESCE(lv.is_live_view, 'N') AS live_view_yn
FROM user u
-- 유저가 디바이스가 없어도 조회
LEFT JOIN user_building ub ON u.id = ub.user_id
LEFT JOIN active_device ad ON ub.id = ad.building_id
LEFT JOIN contract c ON ad.contract_id = c.id
LEFT JOIN latest_user_app_info_cte app ON u.login_id = app.login_id
LEFT JOIN latest_so_version so ON ad.id = so.device_id

LEFT JOIN (
    SELECT user_id, 'Y' AS is_live_view
    FROM history_streaming
    GROUP BY user_id
) lv ON u.login_id = lv.user_id
ORDER BY u.login_id;




SELECT COUNT(*) AS linked_total,

       SUM(last_data_date IS NULL OR last_data_date < NOW() - INTERVAL 3 DAY)                       AS no_data_within_3d,
       SUM(last_data_date <  NOW() - INTERVAL 3  DAY AND last_data_date >= NOW() - INTERVAL 7  DAY) AS off_3_7d,
       SUM(last_data_date <  NOW() - INTERVAL 7  DAY AND last_data_date >= NOW() - INTERVAL 15 DAY) AS off_7_15d,
       SUM(last_data_date <  NOW() - INTERVAL 15 DAY AND last_data_date >= NOW() - INTERVAL 30 DAY) AS off_15_30d,
       SUM(last_data_date <  NOW() - INTERVAL 30 DAY)                                               AS off_over_30d,
       SUM(last_data_date IS NULL)                                                                  AS off_never
FROM (
  SELECT d.serial, MAX(so.create_date) AS last_data_date
  FROM device d
  LEFT JOIN statistics_operation so ON so.device_id = d.id
  WHERE d.linked_yn = 'Y' AND d.device_type = 'ABT'
  GROUP BY d.serial
) t;
