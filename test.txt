WITH latest_device AS (
    SELECT serial, id, device_name, device_type
    FROM (
        SELECT
            serial,
            id,
            device_name,
            device_type,
            ROW_NUMBER() OVER (PARTITION BY serial ORDER BY id DESC) AS rn
        FROM device
    ) t
    WHERE rn = 1
),
clean_summary AS (
    SELECT
        serial,
        SUM(CASE WHEN clean_type_code = 'CLMD0001' THEN 1 ELSE 0 END) AS 고정청정,
        SUM(CASE WHEN clean_type_code = 'CLMD0002' THEN 1 ELSE 0 END) AS 전체공간청정,
        SUM(CASE WHEN clean_type_code = 'CLMD0003' AND call_aqm_type = 'NONE' THEN 1 ELSE 0 END) AS 선택공간청정,
        SUM(CASE WHEN clean_type_code = 'CLMD0003' AND call_aqm_type = 'AUTO' THEN 1 ELSE 0 END) AS 자동호출청정,
        COUNT(*) AS 총청정횟수
    FROM history_clean_result
    WHERE create_date BETWEEN @start_date AND @end_date
    GROUP BY serial
),
llm_summary AS (
    SELECT
        device_id,
        COUNT(*) AS LLM사용
    FROM cloud_llm_usages
    WHERE create_date BETWEEN @start_date AND @end_date
    GROUP BY device_id
),
vs_summary AS (
    SELECT
        deviceSerialNo AS serial,
        COUNT(*) AS 바이탈사인
    FROM vital_sign
    WHERE create_date BETWEEN @start_date AND @end_date
    GROUP BY deviceSerialNo
),
error_summary AS (
    SELECT
        serial,
        COUNT(*) AS 에러횟수
    FROM history_error
    WHERE create_date BETWEEN @start_date AND @end_date
    GROUP BY serial
),
streaming_summary AS (
    SELECT
        serial,
        COUNT(
            CASE
                WHEN streaming_status IN ('STOPPED', 'UNEXPECTED_CLOSE')
                THEN 1
            END
        ) AS 스트리밍사용
    FROM history_streaming
    WHERE requested_at BETWEEN @start_date AND @end_date
    GROUP BY serial
),
follow_summary AS (
    SELECT
        serial,
        COUNT(*) AS 팔로미
    FROM history_status
    WHERE status_type = 'FOLLOW_ME'
      AND status = 'end'
      AND create_date BETWEEN @start_date AND @end_date
    GROUP BY serial
)
SELECT
    d.serial,
    d.device_name,
    d.device_type,
    COALESCE(cs.고정청정, 0) AS 고정청정,
    COALESCE(cs.전체공간청정, 0) AS 전체공간청정,
    COALESCE(cs.선택공간청정, 0) AS 선택공간청정,
    COALESCE(cs.자동호출청정, 0) AS 자동호출청정,
    COALESCE(cs.총청정횟수, 0) AS 총청정횟수,
    COALESCE(ls.LLM사용, 0) AS LLM사용,
    COALESCE(vs.바이탈사인, 0) AS 바이탈사인,
    COALESCE(es.에러횟수, 0) AS 에러횟수,
    COALESCE(ss.스트리밍사용, 0) AS 스트리밍사용,
    COALESCE(fs.팔로미, 0) AS 팔로미
FROM latest_device d
LEFT JOIN clean_summary cs ON d.serial = cs.serial
LEFT JOIN llm_summary ls ON d.id = ls.device_id
LEFT JOIN vs_summary vs ON d.serial = vs.serial
LEFT JOIN error_summary es ON d.serial = es.serial
LEFT JOIN streaming_summary ss ON d.serial = ss.serial
-- 콜레이션 불일치 해결을 위한 COLLATE 설정 추가
LEFT JOIN follow_summary fs
    ON d.serial = fs.serial COLLATE utf8mb4_unicode_ci
WHERE
    COALESCE(cs.총청정횟수, 0)
    + COALESCE(ls.LLM사용, 0)
    + COALESCE(vs.바이탈사인, 0)
    + COALESCE(es.에러횟수, 0)
    + COALESCE(ss.스트리밍사용, 0)
    + COALESCE(fs.팔로미, 0) > 0
